  STRUCT S
----
  signed int i
----
  float f
----
  *signed int pi
----
  *float pf
----
  signed int[20] arr
----
  STRUCT S s
----
  *STRUCT S ps
----
  FUN void testTernary 
    #0: signed int cond
  BEGIN
    *cond ? *i : *i
    *cond ? (float)(*i) : *f
    *cond ? (*signed int)(*i) : *pi
    *cond ? (*float)(*i) : *pf
    *cond ? (*signed int)(*i) : (*signed int)(*arr)
    *cond ? *i : *s
    *cond ? (*STRUCT S)(*i) : *ps
    *cond ? *f : (float)(*i)
    *cond ? *f : *f
    *cond ? *f : *pi
    *cond ? *f : *pf
    *cond ? *f : *arr
    *cond ? *f : *s
    *cond ? *f : *ps
    *cond ? *pi : (*signed int)(*i)
    *cond ? *pi : *f
    *cond ? *pi : *pi
    *cond ? *pi : (*signed int)(*pf)
    *cond ? *pi : (*signed int)(*arr)
    *cond ? *pi : *s
    *cond ? *pi : (*signed int)(*ps)
    *cond ? *pf : (*float)(*i)
    *cond ? *pf : *f
    *cond ? *pf : (*float)(*pi)
    *cond ? *pf : *pf
    *cond ? *pf : (*float)(*arr)
    *cond ? *pf : *s
    *cond ? *pf : (*float)(*ps)
    *cond ? (*signed int)(*arr) : (*signed int)(*i)
    *cond ? *arr : *f
    *cond ? (*signed int)(*arr) : *pi
    *cond ? (*signed int)(*arr) : (*signed int)(*pf)
    *cond ? (*signed int)(*arr) : (*signed int)(*arr)
    *cond ? *arr : *s
    *cond ? (*signed int)(*arr) : (*signed int)(*ps)
    *cond ? *s : *i
    *cond ? *s : *f
    *cond ? *s : *pi
    *cond ? *s : *pf
    *cond ? *s : *arr
    *cond ? *s : *s
    *cond ? *s : *ps
    *cond ? *ps : (*STRUCT S)(*i)
    *cond ? *ps : *f
    *cond ? *ps : (*STRUCT S)(*pi)
    *cond ? *ps : (*STRUCT S)(*pf)
    *cond ? *ps : (*STRUCT S)(*arr)
    *cond ? *ps : *s
    *cond ? *ps : *ps    
  END

MachineFunction 'scp_const_switch' [target = riscv64]
VRegs:
  %v0 : gp/4 ; %11
  %v1 : gp/4 ; %3
  %v2 : gp/4 ; %6
  %v3 : gp/4 ; %30
Frame: <empty>
MBB #0, '<entry>', ir #0, -> #2
  %v0 = li.4 2 ; %11
  %v1 = MOP_COPY.4 $a0 ; %3
  %v2 = MOP_COPY.4 $a1 ; %6

MBB #2, '<FIRST>', ir #2, <- #0, -> #4

MBB #4, '<case_block>', ir #7, <- #2, -> #3
  %v3 = mulw.4 %v1, %v2 ; %30

MBB #3, '<switch_exit>', ir #3, <- #4, -> #1

MBB #1, '<exit>', ir #1, <- #3
  $a0 = MOP_COPY.4 %v3 ; %38
  ret ; %38
